using System;
namespace ConsoleApp6
{
    internal class Program
    {
        static void Main(string[] args)
        {
            int[] numeros = new int[5];
            int opcion;

            do
            {
                Console.WriteLine("\n===== MENÚ =====");
                Console.WriteLine("1 - Cargar el vector");
                Console.WriteLine("2 - Mostrar el vector");
                Console.WriteLine("3 - Buscar un número");
                Console.WriteLine("4 - Sumar todos los elementos");
                Console.WriteLine("5 - Salir");
                Console.Write("Seleccione una opción: ");
                opcion = Convert.ToInt32(Console.ReadLine());

                if (opcion == 1)
                {
                    for (int i = 0; i < numeros.Length; i++)
                    {
                        Console.Write($"Ingrese el número {i + 1}: ");
                        numeros[i] = Convert.ToInt32(Console.ReadLine());
                    }
                }
                else if (opcion == 2)
                {
                    Console.WriteLine("Contenido del vector:");
                    foreach (int n in numeros)
                    {
                        Console.WriteLine(n); 
                    }
                }
                else if (opcion == 3)
                {
                    Console.Write("Ingrese el número a buscar: ");
                    int buscado = Convert.ToInt32(Console.ReadLine());
                    bool encontrado = false;

                    for (int i = 0; i < numeros.Length; i++)
                    {
                        if (numeros[i] == buscado)
                        {
                            Console.WriteLine($"Número encontrado en la posición {i}.");
                            encontrado = true;
                            break;
                        }
                    }

                    if (!encontrado)
                    {
                        Console.WriteLine("Número no encontrado.");
                    }
                }
                else if (opcion == 4)
                {
                    int suma = 0;
                    foreach (int n in numeros)
                    {
                        suma += n;
                    }
                    Console.WriteLine($"La suma de los elementos es: {suma}");
                }
                else if (opcion == 5)
                {
                    Console.WriteLine("Saliendo del programa...");
                }
                else
                {
                    Console.WriteLine("Opción inválida.");
                }

            } while (opcion != 5);

        }
    }
}
