using System;
namespace ConsoleApp6
{
    internal class Program
    {
        static void Main(string[] args)
        {
            string[] ciudades = { "Madrid", "Buenos Aires", "Bogotá", "Lima", "Santiago" };

            Console.Write("Ingrese el nombre de una ciudad: ");
            string ciudadBuscada = Console.ReadLine();

            bool encontrada = false;

            for (int i = 0; i < ciudades.Length; i++)
            {
                if (ciudades[i].ToLower() == ciudadBuscada.ToLower())
                {
                    Console.WriteLine($"Ciudad encontrada en la posición {i + 1}");
                    encontrada = true;
                    break;
                }
            }

            if (!encontrada)
            {
                Console.WriteLine("Ciudad no encontrada.");
            }
        }
    }
}
