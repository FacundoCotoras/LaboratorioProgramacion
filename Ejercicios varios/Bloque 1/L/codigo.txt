using System;
namespace ConsoleApp1
{
    internal class Program
    {
        static void Main(string[] args)
        {
            int suma = 0;

            while (suma <= 100)
            {
                Console.Write("Ingrese un numero: ");
                suma = suma + Convert.ToInt16(Console.ReadLine());
            }

            Console.WriteLine("Total acumulado: " + suma);
        }
    }
}
