using System;
namespace ConsoleApp1
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Console.Write("Ingrese un numero: ");
            int numero = Convert.ToInt16(Console.ReadLine());
            for (int i = 0; i < 13; i++)
            {
                Console.WriteLine($"{i}x{numero} = {i * numero}");
            }
        }
    }
}
