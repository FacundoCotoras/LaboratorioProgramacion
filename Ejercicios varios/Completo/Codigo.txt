        static void SistemaNombres()
        {
            string[] nombres = new string[100];
            int cantidad = 0;
            bool repetido = false;

            Console.WriteLine("Ingrese nombres (se detiene cuando se repita alguno):");

            while (!repetido)
            {
                string nombre = Console.ReadLine();


                for (int i = 0; i < cantidad; i++)
                {
                    if (nombres[i].ToLower() == nombre.ToLower())
                    {
                        repetido = true;
                        break;
                    }
                }

                if (!repetido)
                {
                    nombres[cantidad] = nombre;
                    cantidad++;
                }
            }

            Console.WriteLine($"Se ingresaron {cantidad} nombres antes del duplicado.");
        }

        static void PiedraPapelTijera()
        {
            int victoriasUsuario = 0;
            int victoriasPC = 0;
            Random rnd = new Random();

            Console.WriteLine("Juego: Piedra, Papel o Tijera (3 rondas)");

            for (int ronda = 1; ronda <= 3; ronda++)
            {
                Console.WriteLine($"Ronda {ronda}:");

                string usuario;
                do
                {
                    Console.Write("Elija (piedra, papel o tijera): ");
                    usuario = Console.ReadLine().ToLower();

                    if (usuario != "piedra" && usuario != "papel" && usuario != "tijera")
                    {
                        Console.WriteLine("Entrada inválida. Debe ser piedra, papel o tijera.");
                    }

                } while (usuario != "piedra" && usuario != "papel" && usuario != "tijera");

                int jugadaPC = rnd.Next(3);
                string pc;

                if (jugadaPC == 0)
                    pc = "piedra";
                else if (jugadaPC == 1)
                    pc = "papel";
                else
                    pc = "tijera";

                Console.WriteLine($"La PC eligió: {pc}");

                if (usuario == pc)
                {
                    Console.WriteLine("Empate.");
                }
                else if ((usuario == "piedra" && pc == "tijera") ||
                         (usuario == "papel" && pc == "piedra") ||
                         (usuario == "tijera" && pc == "papel"))
                {
                    Console.WriteLine("Ganaste esta ronda.");
                    victoriasUsuario++;
                }
                else
                {
                    Console.WriteLine("La PC ganó esta ronda.");
                    victoriasPC++;
                }
            }

            Console.WriteLine("Resultado final:");
            if (victoriasUsuario > victoriasPC)
                Console.WriteLine($"Ganaste más rondas ({victoriasUsuario} vs {victoriasPC}).");
            else if (victoriasPC > victoriasUsuario)
                Console.WriteLine($"La PC ganó más rondas ({victoriasPC} vs {victoriasUsuario}).");
            else
                Console.WriteLine($"Empate total ({victoriasUsuario} vs {victoriasPC}).");
        }

                Console.WriteLine("Ganó la Opción B.");
            else if (opcion3 > opcion1 && opcion3 > opcion2)
                Console.WriteLine("Ganó la Opción C.");
            else
                Console.WriteLine("Hubo empate.");

        static void CuentaBancaria()
        {
            int saldo = 1000;
            string opcion;

            Console.WriteLine("Bienvenido a tu cuenta bancaria.");
            Console.WriteLine("Saldo inicial: $1000");

            do
            {
                Console.WriteLine("Opciones: depositar, retirar, salir");
                Console.Write("Ingrese una opción: ");
                opcion = Console.ReadLine().ToLower();

                if (opcion == "depositar")
                {
                    Console.Write("Ingrese el monto a depositar: ");
                    int monto = Convert.ToInt32(Console.ReadLine());
                    saldo += monto;
                    Console.WriteLine($"Depósito realizado. Saldo actual: ${saldo}");
                }
                else if (opcion == "retirar")
                {
                    Console.Write("Ingrese el monto a retirar: ");
                    int monto = Convert.ToInt32(Console.ReadLine());

                    if (monto <= saldo)
                    {
                        saldo -= monto;
                        Console.WriteLine($"Retiro realizado. Saldo actual: ${saldo}");
                    }
                    else
                    {
                        Console.WriteLine("No puede retirar más de lo que tiene.");
                    }
                }
                else if (opcion != "salir")
                {
                    Console.WriteLine("Opción inválida. Intente nuevamente.");
                }

            } while (opcion != "salir");

            Console.WriteLine($"Operación finalizada. Saldo final: ${saldo}");
        }
        static void Divisibles2o5()
        {

            int numero = 1;

            Console.Write("Primeros 50 números divisibles por 2 o por 5, pero no por ambos:");

            while (contador < 50)
            {
                bool divisiblePor2 = numero % 2 == 0;
                bool divisiblePor5 = numero % 5 == 0;
                if ((divisiblePor2 && !divisiblePor5) || (!divisiblePor2 && divisiblePor5))

                    Console.Write($" -{numero}-");
                    contador++;
                }

                numero++;
            }
        }

        static void InvertirPalabras()
        {
            Console.Write("Ingrese una frase: ");
            string frase = Console.ReadLine();
            string[] palabras = frase.Split(' ');
            string resultado = "";

            for (int i = 0; i < palabras.Length; i++)
            {
                string invertida = "";
                for (int j = palabras[i].Length - 1; j >= 0; j--)
                {
                    invertida += palabras[i][j];
                }

                resultado += invertida;

                if (i < palabras.Length - 1)
                {
                    resultado += " ";
                }
            }

            Console.WriteLine(resultado);
        }

            string palabra = Console.ReadLine().ToLower();

            for (int i = 0; i < palabra.Length; i++)
            {
                char letra = palabra[i];
                bool repetida = false;

                for (int k = 0; k < i; k++)
                {
                    if (palabra[k] == letra)
                    {
                        repetida = true;
                        break;
                    }
                }

                if (!repetida)
                {
                    int contador = 0;
                    for (int j = 0; j < palabra.Length; j++)
                    {
                        if (palabra[j] == letra)
                        {
                            contador++;
                        }
                    }
                    Console.WriteLine($"La letra '{letra}' aparece {contador} veces.");
                }
            }
        }

        static void CalcularIMC()
        {

            decimal peso = Convert.ToDecimal(Console.ReadLine());

            Console.Write("Ingrese su altura en metros: ");
            decimal altura = Convert.ToDecimal(Console.ReadLine());

            decimal imc = peso / (altura * altura);
            decimal imcRedondeado = Math.Round(imc, 2);

            Console.WriteLine($"Su IMC es: {imcRedondeado}");

            if (imc < 18.5m)
            {
                Console.WriteLine("Categoría: Bajo peso");
            }
            else if (imc >= 18.5m && imc < 25m)
            {
                Console.WriteLine("Categoría: Normal");
            }
            else if (imc >= 25m && imc < 30m)
            {
                Console.WriteLine("Categoría: Sobrepeso");
            }
            else
            {
                Console.WriteLine("Categoría: Obesidad");
            }
        }
